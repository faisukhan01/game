    22→    22→/** Deterministic 32-bit RNG (mulberry32). */
    23→    23→function rng32(seed: number): () => number {
    24→    24→  let a = seed >>> 0;
    25→    25→  return () => {
    26→    26→    a = (a + 0x6d2b79f5) >>> 0;
    27→    27→    let t = a;
    28→    28→    t = Math.imul(t ^ (t >>> 15), t | 1);
    29→    29→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    30→    30→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    31→    31→  };
    32→    32→}

   147→   147→function bakeSky(): HTMLCanvasElement {
   148→   148→  const [c, ctx] = makeCanvas(WORLD_W, ENV_HORIZON);
   149→   149→  const W = WORLD_W;
   150→   150→  const H = ENV_HORIZON;
   151→   151→
   152→   152→  // Dusk gradient.
   153→   153→  const g = ctx.createLinearGradient(0, 0, 0, H);
   154→   154→  g.addColorStop(0, C.skyTop);
   155→   155→  g.addColorStop(0.45, "#4e3a2e");
   156→   156→  g.addColorStop(0.78, C.skyMid);
   157→   157→  g.addColorStop(1, C.skyGlow);
   158→   158→  ctx.fillStyle = g;
   159→   159→  ctx.fillRect(0, 0, W, H);
   160→   160→
   161→   161→  // Low sun with a soft halo.
   162→   162→  const sunX = 1160;
   163→   163→  const sunY = 168;
   164→   164→  const halo = ctx.createRadialGradient(sunX, sunY, 8, sunX, sunY, 190);
   165→   165→  halo.addColorStop(0, "rgba(255,233,187,0.85)");
   166→   166→  halo.addColorStop(0.25, "rgba(240,190,120,0.38)");
   167→   167→  halo.addColorStop(1, "rgba(240,190,120,0)");
   168→   168→  ctx.fillStyle = halo;
   169→   169→  ctx.fillRect(sunX - 200, sunY - 200, 400, 400);
   170→   170→  ctx.fillStyle = C.skySun;
   171→   171→  ctx.beginPath();
   172→   172→  ctx.arc(sunX, sunY, 26, 0, Math.PI * 2);
   173→   173→  ctx.fill();
   174→   174→
   175→   175→  const rand = rng32(0xC0FFEE);
   176→   176→
   177→   177→  // ---- far skyline: derelict society towers, half-demolished --------------
   178→   178→  const drawBlock = (
   179→   179→    bx: number, bw: number, bh: number, color: string, seed: number,
   180→   180→  ): void => {
   181→   181→    const r = rng32(seed);
   182→   182→    const top = H - bh;
   183→   183→    ctx.fillStyle = color;
   184→   184→    ctx.fillRect(bx, top, bw, bh);
   185→   185→    // Broken roofline — nibble the parapet.
   186→   186→    ctx.fillStyle = C.skyTop;
   187→   187→    let nx = bx;
   188→   188→    while (nx < bx + bw) {
   189→   189→      if (r() < 0.55) {
   190→   190→        const nw = 6 + r() * 16;
   191→   191→        const nd = 4 + r() * 10;
   192→   192→        ctx.fillRect(nx, top, nw, nd);
   193→   193→        nx += nw;
   194→   194→      } else nx += 8 + r() * 20;
   195→   195→    }
   196→   196→    // Water tank on some roofs.
   197→   197→    if (r() < 0.5) {
   198→   198→      const tx = bx + 8 + r() * (bw - 26);
   199→   199→      ctx.fillStyle = color;
   200→   200→      ctx.fillRect(tx, top - 12, 14, 12);
   201→   201→      ctx.fillRect(tx - 2, top - 14, 18, 3);
   202→   202→    }
   203→   203→    // A few faint warm windows (most abandoned, some still lit).
   204→   204→    for (let wy = top + 10; wy < H - 8; wy += 14) {
   205→   205→      for (let wx = bx + 6; wx < bx + bw - 8; wx += 12) {
   206→   206→        const lit = r();
   207→   207→        if (lit < 0.06) ctx.fillStyle = "rgba(255,190,110,0.30)";
   208→   208→        else if (lit < 0.2) ctx.fillStyle = "rgba(0,0,0,0.35)";
   209→   209→        else continue;
   210→   210→        ctx.fillRect(wx, wy, 6, 7);
   211→   211→      }
   212→   212→    }
   213→   213→  };
   214→   214→
   215→   215→  // Far band (behind, lighter).
   216→   216→  let x = -20;
   217→   217→  let i = 0;
   218→   218→  while (x < W + 20) {
   219→   219→    const bw = 46 + rand() * 90;
   220→   220→    drawBlock(x, bw, 52 + rand() * 84, C.skylineFar, 0x5157 + i * 97);
   221→   221→    x += bw + 4 + rand() * 26;
   222→   222→    i++;
   223→   223→  }
   224→   224→  // Near band (darker, sparser, taller).
   225→   225→  x = 30;
   226→   226→  i = 0;
   227→   227→  while (x < W + 20) {
   228→   228→    const bw = 60 + rand() * 110;
   229→   229→    drawBlock(x, bw, 30 + rand() * 58, C.skylineNear, 0x9A21 + i * 131);
   230→   230→    x += bw + 30 + rand() * 90;
   231→   231→    i++;
   232→   232→  }
   233→   233→
   234→   234→  // Power poles with sagging wires across the horizon.
   235→   235→  const poles = [90, 420, 760, 1080, 1420];
   236→   236→  ctx.strokeStyle = "#241d19";
   237→   237→  ctx.lineWidth = 3;
   238→   238→  for (const px of poles) {
   239→   239→    const ph = 34 + (px % 37);
   240→   240→    ctx.beginPath();
   241→   241→    ctx.moveTo(px, H);
   242→   242→    ctx.lineTo(px, H - ph);
   243→   243→    ctx.moveTo(px - 9, H - ph + 7);
   244→   244→    ctx.lineTo(px + 9, H - ph + 7);
   245→   245→    ctx.stroke();
   246→   246→  }
   247→   247→  ctx.strokeStyle = "rgba(30,24,20,0.8)";
   248→   248→  ctx.lineWidth = 1.2;
   249→   249→  ctx.beginPath();
   250→   250→  for (let p = 0; p < poles.length - 1; p++) {
   251→   251→    const x1 = poles[p];
   252→   252→    const x2 = poles[p + 1];
   253→   253→    const y1 = H - (34 + (x1 % 37)) + 7;
   254→   254→    const y2 = H - (34 + (x2 % 37)) + 7;
   255→   255→    ctx.moveTo(x1, y1);
   256→   256→    ctx.quadraticCurveTo((x1 + x2) / 2, Math.max(y1, y2) + 14, x2, y2);
   257→   257→  }
   258→   258→  ctx.stroke();
   259→   259→
   260→   260→  // Dead scrub on the horizon line.
   261→   261→  for (let sx = 0; sx < W; sx += 14) {
   262→   262→    if (rand() < 0.4) continue;
   263→   263→    const h = 3 + rand() * 9;
   264→   264→    ctx.strokeStyle = "rgba(40,32,24,0.7)";
   265→   265→    ctx.lineWidth = 1.4;
   266→   266→    ctx.beginPath();
   267→   267→    ctx.moveTo(sx, H);
   268→   268→    ctx.lineTo(sx + (rand() - 0.5) * 8, H - h);
   269→   269→    ctx.stroke();
   270→   270→  }
   271→   271→  return c;
   272→   272→}

   276→   276→function bakeGround(): HTMLCanvasElement {
   277→   277→  const [c, ctx] = makeCanvas(WORLD_W, GROUND_H);
   278→   278→  const W = WORLD_W;
   279→   279→  const H = GROUND_H;

   285→   285→  g.addColorStop(0.4, C.grassMid);

   288→   288→  ctx.fillRect(0, 0, W, H);

   290→   290→  // Large soft tonal patches (mown / dried out areas).

   293→   293→    const py = rand() * H;
   294→   294→    const pr = 60 + rand() * 150;
   295→   295→    const dark = rand() < 0.5;

   299→   299→    ctx.fillStyle = pg;
   300→   300→    ctx.beginPath();
   301→   301→    ctx.ellipse(px, py, pr, pr * 0.55, 0, 0, Math.PI * 2);
   302→   302→    ctx.fill();
   303→   303→  }
   304→   304→
   305→   305→  // Dirt worn patches.
   306→   306→  for (let p = 0; p < 22; p++) {
   307→   307→    const px = rand() * W;
   308→   308→    const py = rand() * H;
   309→   309→    const pr = 24 + rand() * 60;
   310→   310→    ctx.fillStyle = "rgba(107,90,66,0.5)";
   311→   311→    ctx.beginPath();
   312→   312→    ctx.ellipse(px, py, pr, pr * 0.5, rand() * Math.PI, 0, Math.PI * 2);
   313→   313→    ctx.fill();
   314→   314→    ctx.fillStyle = "rgba(125,108,82,0.35)";
   315→   315→    ctx.beginPath();
   316→   316→    ctx.ellipse(px - pr * 0.15, py - pr * 0.1, pr * 0.6, pr * 0.3, 0, 0, Math.PI * 2);
   317→   317→    ctx.fill();
   318→   318→  }
   319→   319→
   320→   320→  // Cracked asphalt road band across the lower half.
   321→   321→  const roadTop = localY(540);
   322→   322→  const roadBot = localY(668);
   323→   323→  ctx.fillStyle = C.road;
   324→   324→  ctx.fillRect(0, roadTop, W, roadBot - roadTop);
   325→   325→  // Edge wear.
   326→   326→  ctx.fillStyle = C.roadEdge;
   327→   327→  ctx.fillRect(0, roadTop, W, 4);
   328→   328→  ctx.fillRect(0, roadBot - 4, W, 4);
   329→   329→  ctx.fillStyle = "rgba(76,80,57,0.9)"; // grass creeping over the edges
   330→   330→  for (let ex = 0; ex < W; ex += 10) {
   331→   331→    if (rand() < 0.55) ctx.fillRect(ex, roadTop - 2 - rand() * 3, 5 + rand() * 6, 3);
   332→   332→    if (rand() < 0.55) ctx.fillRect(ex, roadBot - 1 + rand() * 2, 5 + rand() * 6, 3);
   333→   333→  }
   334→   334→  // Faded center dashes.
   335→   335→  ctx.fillStyle = C.roadDash;
   336→   336→  for (let dx = 20; dx < W; dx += 120) {
   337→   337→    ctx.globalAlpha = 0.35 + rand() * 0.3;
   338→   338→    ctx.fillRect(dx, (roadTop + roadBot) / 2 - 3, 52, 5);
   339→   339→  }
   340→   340→  ctx.globalAlpha = 1;
   341→   341→  // Potholes + tar patches + cracks.
   342→   342→  for (let p = 0; p < 16; p++) {
   343→   343→    const px = rand() * W;
   344→   344→    const py = roadTop + 8 + rand() * (roadBot - roadTop - 16);
   345→   345→    ctx.fillStyle = "rgba(24,26,28,0.55)";
   346→   346→    ctx.beginPath();
   347→   347→    ctx.ellipse(px, py, 7 + rand() * 16, 3 + rand() * 6, 0, 0, Math.PI * 2);
   348→   348→    ctx.fill();
   349→   349→  }
   350→   350→  ctx.strokeStyle = "rgba(20,22,24,0.5)";
   351→   351→  ctx.lineWidth = 1.2;
   352→   352→  ctx.beginPath();
   353→   353→  for (let k = 0; k < 26; k++) {
   354→   354→    let cx = rand() * W;
   355→   355→    let cy = roadTop + rand() * (roadBot - roadTop);
   356→   356→    ctx.moveTo(cx, cy);
   357→   357→    for (let s = 0; s < 4; s++) {
   358→   358→      cx += (rand() - 0.5) * 34;
   359→   359→      cy += (rand() - 0.5) * 12;
   360→   360→      ctx.lineTo(cx, cy);
   361→   361→    }
   362→   362→  }
   363→   363→  ctx.stroke();
   364→   364→
   365→   365→  // Dirt footpath: from the road up to the restaurant door, plus a rim path.
   366→   366→  const pathW = 46;
   367→   367→  const drawPath = (x0: number, y0: number, x1: number, y1: number): void => {
   368→   368→    const steps = 22;
   369→   369→    ctx.fillStyle = "rgba(138,119,88,0.85)";
   370→   370→    for (let s = 0; s <= steps; s++) {
   371→   371→      const t = s / steps;
   372→   372→      const px = x0 + (x1 - x0) * t + Math.sin(t * 5) * 14;
   373→   373→      const py = y0 + (y1 - y0) * t;
   374→   374→      const w = pathW * (0.75 + 0.25 * Math.sin(t * 9));
   375→   375→      ctx.beginPath();
   376→   376→      ctx.ellipse(px, py, w / 2, (w / 2) * TILT * 0.9, 0, 0, Math.PI * 2);
   377→   377→      ctx.fill();
   378→   378→    }
   379→   379→  };
   380→   380→  drawPath(800, roadBot + 6, 800, localY(492)); // road → restaurant front
   381→   381→  drawPath(120, localY(300), 380, localY(320)); // rim path upper-left
   382→   382→  drawPath(1420, localY(320), 1560, localY(350));
   383→   383→  drawPath(240, roadTop - 4, 500, localY(500)); // road → upper park
   384→   384→  // Path pebbles.
   385→   385→  for (let p = 0; p < 90; p++) {
   386→   386→    const t = rand();
   387→   387→    const px = 800 + Math.sin(t * 5) * 14 + (rand() - 0.5) * pathW;
   388→   388→    const py = localY(492) + t * (roadBot + 6 - localY(492));
   389→   389→    ctx.fillStyle = "rgba(160,144,116,0.5)";
   390→   390→    ctx.fillRect(px, py, 2 + rand() * 2, 1.5);
   391→   391→  }
   392→   392→
   393→   393→  // Grass tufts — three quick strokes each.
   394→   394→  const tuftColors = ["#5a6440", "#6e7a4a", "#49542f", "#77824e"];
   395→   395→  for (let t = 0; t < 620; t++) {
   396→   396→    const px = rand() * W;
   397→   397→    const py = rand() * H;
   398→   398→    if (py > roadTop - 4 && py < roadBot + 4) continue;
   399→   399→    ctx.strokeStyle = tuftColors[(rand() * tuftColors.length) | 0];
   400→   400→    ctx.lineWidth = 1.3;
   401→   401→    ctx.beginPath();
   402→   402→    for (let b = 0; b < 3; b++) {
   403→   403→      const bx = px + (b - 1) * 2.4;

   941→   941→function drawTree(ctx: CanvasRenderingContext2D, d: Decor, t: number): void {
   942→   942→  const ax = d.x;
   943→   943→  const ay = ENV_HORIZON + localY(d.y);
   944→   944→  const S = 52 * d.scale;
   945→   945→  const sway = Math.sin(t * 1.1 + d.variant * 1.7) * 0.028 + Math.sin(t * 2.3 + d.variant) * 0.012;
   946→   946→
   947→   947→  // Shadow.
   948→   948→  ctx.fillStyle = "rgba(10,12,8,0.4)";
   949→   949→  ctx.beginPath();
   950→   950→  ctx.ellipse(ax + 6, ay + 3, S * 0.55, S * 0.14, 0, 0, Math.PI * 2);
   951→   951→  ctx.fill();
   952→   952→
   953→   953→  const lean = sway * S;
   954→   954→  if (d.dead) {
   955→   955→    // Bare dead tree — recursive-ish branches.
   956→   956→    ctx.strokeStyle = C.deadWood;
   957→   957→    ctx.lineCap = "round";
   958→   958→    ctx.lineWidth = S * 0.09;
   959→   959→    ctx.beginPath();
   960→   960→    ctx.moveTo(ax, ay);
   961→   961→    const topX = ax + lean;
   962→   962→    const topY = ay - S * 1.75;
   963→   963→    ctx.quadraticCurveTo(ax + S * 0.06, ay - S * 0.9, topX, topY);
   964→   964→    ctx.stroke();
   965→   965→    const branch = (
   966→   966→      x0: number, y0: number, ang: number, len: number, w: number, depth: number,
   967→   967→    ): void => {
   968→   968→      if (depth === 0 || len < 4) return;
   969→   969→      const x1 = x0 + Math.cos(ang) * len;
   970→   970→      const y1 = y0 + Math.sin(ang) * len;
   971→   971→      ctx.lineWidth = w;
   972→   972→      ctx.beginPath();
   973→   973→      ctx.moveTo(x0, y0);
   974→   974→      ctx.lineTo(x1, y1);
   975→   975→      ctx.stroke();
   976→   976→      branch(x1, y1, ang - 0.42 - sway * 0.4, len * 0.68, w * 0.6, depth - 1);
   977→   977→      branch(x1, y1, ang + 0.38 + sway * 0.4, len * 0.66, w * 0.6, depth - 1);
   978→   978→    };
   979→   979→    const tipX = ax + lean;
   980→   980→    const tipY = ay - S * 1.75;
   981→   981→    branch(tipX, tipY + S * 0.35, -Math.PI / 2 - 0.65 + sway, S * 0.5, S * 0.06, 3);
   982→   982→    branch(tipX, tipY + S * 0.55, -Math.PI / 2 + 0.6 + sway, S * 0.55, S * 0.07, 3);
   983→   983→    branch(tipX, tipY + S * 0.15, -Math.PI / 2 + 0.18 + sway * 0.5, S * 0.4, S * 0.05, 2);
   984→   984→    return;
   985→   985→  }
   986→   986→
   987→   987→  // Living tree: trunk + layered canopy blobs.
   988→   988→  ctx.strokeStyle = "#3e332a";
   989→   989→  ctx.lineCap = "round";
   990→   990→  ctx.lineWidth = S * 0.13;
   991→   991→  ctx.beginPath();
   992→   992→  ctx.moveTo(ax, ay);
   993→   993→  ctx.quadraticCurveTo(ax + S * 0.05, ay - S * 0.8, ax + lean, ay - S * 1.35);
   994→   994→  ctx.stroke();
   995→   995→  // Branch stubs.
   996→   996→  ctx.lineWidth = S * 0.06;
   997→   997→  ctx.beginPath();
   998→   998→  ctx.moveTo(ax + S * 0.02, ay - S * 0.85);
   999→   999→  ctx.lineTo(ax - S * 0.34 + lean * 0.6, ay - S * 1.15);
  1000→  1000→  ctx.moveTo(ax + S * 0.03, ay - S * 1.0);
  1001→  1001→  ctx.lineTo(ax + S * 0.38 + lean * 0.6, ay - S * 1.28);
  1002→  1002→  ctx.stroke();
  1003→  1003→
  1004→  1004→  const cy = ay - S * 1.62;
  1005→  1005→  const cx = ax + lean * 1.6;
  1006→  1006→  const blobs: [number, number, number][] = [
  1007→  1007→    [0, 0, 0.62],
  1008→  1008→    [-0.42, 0.16, 0.44],
  1009→  1009→    [0.44, 0.12, 0.46],
  1010→  1010→    [-0.18, -0.3, 0.4],
  1011→  1011→    [0.24, -0.26, 0.38],
  1012→  1012→  ];
  1013→  1013→  for (const [ox, oy, r] of blobs) {
  1014→  1014→    ctx.fillStyle = C.leafDark;
  1015→  1015→    ctx.beginPath();
  1016→  1016→    ctx.ellipse(cx + ox * S, cy + oy * S, r * S, r * S * 0.86, 0, 0, Math.PI * 2);
  1017→  1017→    ctx.fill();
  1018→  1018→  }
  1019→  1019→  for (const [ox, oy, r] of blobs) {
  1020→  1020→    ctx.fillStyle = C.leaf;
  1021→  1021→    ctx.beginPath();
  1022→  1022→    ctx.ellipse(cx + ox * S - 1.5, cy + oy * S - 2.5, r * S * 0.9, r * S * 0.76, 0, 0, Math.PI * 2);
  1023→  1023→    ctx.fill();
  1024→  1024→  }
  1025→  1025→  // Sun-side highlights.
  1026→  1026→  for (const [ox, oy, r] of blobs) {
  1027→  1027→    ctx.fillStyle = C.leafLight;
  1028→  1028→    ctx.beginPath();
  1029→  1029→    ctx.ellipse(cx + ox * S + r * S * 0.24, cy + oy * S - r * S * 0.3, r * S * 0.42, r * S * 0.3, 0, 0, Math.PI * 2);
  1030→  1030→    ctx.fill();
  1031→  1031→  }
  1032→  1032→}
