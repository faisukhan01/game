    22→/** Deterministic 32-bit RNG (mulberry32). */
    23→function rng32(seed: number): () => number {
    24→  let a = seed >>> 0;
    25→  return () => {
    26→    a = (a + 0x6d2b79f5) >>> 0;
    27→    let t = a;
    28→    t = Math.imul(t ^ (t >>> 15), t | 1);
    29→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    30→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    31→  };
    32→}

    34→function makeCanvas(w: number, h: number): [HTMLCanvasElement, CanvasRenderingContext2D] {
    35→  const c = document.createElement("canvas");
    36→  c.width = w;
    37→  c.height = h;
    38→  const ctx = c.getContext("2d");
    39→  if (!ctx) throw new Error("2D context unavailable");
    40→  return [c, ctx];
    41→}

   135→function getBaked(): Baked {
   136→  if (baked) return baked;

   138→    sky: bakeSky(),

   147→function bakeSky(): HTMLCanvasElement {
   148→  const [c, ctx] = makeCanvas(WORLD_W, ENV_HORIZON);
   149→  const W = WORLD_W;
   150→  const H = ENV_HORIZON;
   151→
   152→  // Dusk gradient.
   153→  const g = ctx.createLinearGradient(0, 0, 0, H);
   154→  g.addColorStop(0, C.skyTop);
   155→  g.addColorStop(0.45, "#4e3a2e");
   156→  g.addColorStop(0.78, C.skyMid);
   157→  g.addColorStop(1, C.skyGlow);
   158→  ctx.fillStyle = g;
   159→  ctx.fillRect(0, 0, W, H);
   160→
   161→  // Low sun with a soft halo.
   162→  const sunX = 1160;
   163→  const sunY = 168;
   164→  const halo = ctx.createRadialGradient(sunX, sunY, 8, sunX, sunY, 190);
   165→  halo.addColorStop(0, "rgba(255,233,187,0.85)");
   166→  halo.addColorStop(0.25, "rgba(240,190,120,0.38)");
   167→  halo.addColorStop(1, "rgba(240,190,120,0)");
   168→  ctx.fillStyle = halo;
   169→  ctx.fillRect(sunX - 200, sunY - 200, 400, 400);
   170→  ctx.fillStyle = C.skySun;
   171→  ctx.beginPath();
   172→  ctx.arc(sunX, sunY, 26, 0, Math.PI * 2);
   173→  ctx.fill();
   174→
   175→  const rand = rng32(0xC0FFEE);
   176→
   177→  // ---- far skyline: derelict society towers, half-demolished --------------
   178→  const drawBlock = (
   179→    bx: number, bw: number, bh: number, color: string, seed: number,
   180→  ): void => {
   181→    const r = rng32(seed);
   182→    const top = H - bh;
   183→    ctx.fillStyle = color;
   184→    ctx.fillRect(bx, top, bw, bh);
   185→    // Broken roofline — nibble the parapet.
   186→    ctx.fillStyle = C.skyTop;
   187→    let nx = bx;
   188→    while (nx < bx + bw) {
   189→      if (r() < 0.55) {
   190→        const nw = 6 + r() * 16;
   191→        const nd = 4 + r() * 10;
   192→        ctx.fillRect(nx, top, nw, nd);
   193→        nx += nw;
   194→      } else nx += 8 + r() * 20;
   195→    }
   196→    // Water tank on some roofs.
   197→    if (r() < 0.5) {
   198→      const tx = bx + 8 + r() * (bw - 26);
   199→      ctx.fillStyle = color;
   200→      ctx.fillRect(tx, top - 12, 14, 12);
   201→      ctx.fillRect(tx - 2, top - 14, 18, 3);
   202→    }
   203→    // A few faint warm windows (most abandoned, some still lit).
   204→    for (let wy = top + 10; wy < H - 8; wy += 14) {
   205→      for (let wx = bx + 6; wx < bx + bw - 8; wx += 12) {
   206→        const lit = r();
   207→        if (lit < 0.06) ctx.fillStyle = "rgba(255,190,110,0.30)";
   208→        else if (lit < 0.2) ctx.fillStyle = "rgba(0,0,0,0.35)";
   209→        else continue;
   210→        ctx.fillRect(wx, wy, 6, 7);
   211→      }
   212→    }
   213→  };
   214→
   215→  // Far band (behind, lighter).
   216→  let x = -20;
   217→  let i = 0;
   218→  while (x < W + 20) {
   219→    const bw = 46 + rand() * 90;
   220→    drawBlock(x, bw, 52 + rand() * 84, C.skylineFar, 0x5157 + i * 97);
   221→    x += bw + 4 + rand() * 26;
   222→    i++;
   223→  }
   224→  // Near band (darker, sparser, taller).
   225→  x = 30;
   226→  i = 0;
   227→  while (x < W + 20) {
   228→    const bw = 60 + rand() * 110;
   229→    drawBlock(x, bw, 30 + rand() * 58, C.skylineNear, 0x9A21 + i * 131);
   230→    x += bw + 30 + rand() * 90;
   231→    i++;
   232→  }
   233→
   234→  // Power poles with sagging wires across the horizon.
   235→  const poles = [90, 420, 760, 1080, 1420];
   236→  ctx.strokeStyle = "#241d19";
   237→  ctx.lineWidth = 3;
   238→  for (const px of poles) {
   239→    const ph = 34 + (px % 37);
   240→    ctx.beginPath();
   241→    ctx.moveTo(px, H);
   242→    ctx.lineTo(px, H - ph);
   243→    ctx.moveTo(px - 9, H - ph + 7);
   244→    ctx.lineTo(px + 9, H - ph + 7);
   245→    ctx.stroke();
   246→  }
   247→  ctx.strokeStyle = "rgba(30,24,20,0.8)";
   248→  ctx.lineWidth = 1.2;
   249→  ctx.beginPath();
   250→  for (let p = 0; p < poles.length - 1; p++) {
   251→    const x1 = poles[p];
   252→    const x2 = poles[p + 1];
   253→    const y1 = H - (34 + (x1 % 37)) + 7;
   254→    const y2 = H - (34 + (x2 % 37)) + 7;
   255→    ctx.moveTo(x1, y1);
   256→    ctx.quadraticCurveTo((x1 + x2) / 2, Math.max(y1, y2) + 14, x2, y2);
   257→  }
   258→  ctx.stroke();
   259→
   260→  // Dead scrub on the horizon line.
   261→  for (let sx = 0; sx < W; sx += 14) {
   262→    if (rand() < 0.4) continue;
   263→    const h = 3 + rand() * 9;
   264→    ctx.strokeStyle = "rgba(40,32,24,0.7)";
   265→    ctx.lineWidth = 1.4;
   266→    ctx.beginPath();
   267→    ctx.moveTo(sx, H);
   268→    ctx.lineTo(sx + (rand() - 0.5) * 8, H - h);
   269→    ctx.stroke();
   270→  }
   271→  return c;
   272→}

   276→function bakeGround(): HTMLCanvasElement {
   277→  const [c, ctx] = makeCanvas(WORLD_W, GROUND_H);
   278→  const W = WORLD_W;
   279→  const H = GROUND_H;
   280→  const rand = rng32(0x600D13);
   281→
   282→  // Dry park grass — slightly darker toward the camera.
   283→  const g = ctx.createLinearGradient(0, 0, 0, H);
   284→  g.addColorStop(0, C.grassFar);
   285→  g.addColorStop(0.4, C.grassMid);
   286→  g.addColorStop(1, C.grassNear);
   287→  ctx.fillStyle = g;
   288→  ctx.fillRect(0, 0, W, H);
   289→
   290→  // Large soft tonal patches (mown / dried out areas).
   291→  for (let p = 0; p < 34; p++) {
   292→    const px = rand() * W;
   293→    const py = rand() * H;
   294→    const pr = 60 + rand() * 150;
   295→    const dark = rand() < 0.5;
   296→    const pg = ctx.createRadialGradient(px, py, pr * 0.2, px, py, pr);
   297→    pg.addColorStop(0, dark ? "rgba(38,44,28,0.30)" : "rgba(112,112,72,0.22)");
   298→    pg.addColorStop(1, "rgba(0,0,0,0)");
   299→    ctx.fillStyle = pg;
   300→    ctx.beginPath();
   301→    ctx.ellipse(px, py, pr, pr * 0.55, 0, 0, Math.PI * 2);
   302→    ctx.fill();
   303→  }
   304→
   305→  // Dirt worn patches.
   306→  for (let p = 0; p < 22; p++) {
   307→    const px = rand() * W;
   308→    const py = rand() * H;
   309→    const pr = 24 + rand() * 60;
   310→    ctx.fillStyle = "rgba(107,90,66,0.5)";
   311→    ctx.beginPath();
   312→    ctx.ellipse(px, py, pr, pr * 0.5, rand() * Math.PI, 0, Math.PI * 2);
   313→    ctx.fill();
   314→    ctx.fillStyle = "rgba(125,108,82,0.35)";
   315→    ctx.beginPath();
   316→    ctx.ellipse(px - pr * 0.15, py - pr * 0.1, pr * 0.6, pr * 0.3, 0, 0, Math.PI * 2);
   317→    ctx.fill();
   318→  }
   319→
   320→  // Cracked asphalt road band across the lower half.
   321→  const roadTop = localY(540);
   322→  const roadBot = localY(668);
   323→  ctx.fillStyle = C.road;
   324→  ctx.fillRect(0, roadTop, W, roadBot - roadTop);
   325→  // Edge wear.
   326→  ctx.fillStyle = C.roadEdge;
   327→  ctx.fillRect(0, roadTop, W, 4);
   328→  ctx.fillRect(0, roadBot - 4, W, 4);
   329→  ctx.fillStyle = "rgba(76,80,57,0.9)"; // grass creeping over the edges
   330→  for (let ex = 0; ex < W; ex += 10) {
   331→    if (rand() < 0.55) ctx.fillRect(ex, roadTop - 2 - rand() * 3, 5 + rand() * 6, 3);
   332→    if (rand() < 0.55) ctx.fillRect(ex, roadBot - 1 + rand() * 2, 5 + rand() * 6, 3);
   333→  }
   334→  // Faded center dashes.
   335→  ctx.fillStyle = C.roadDash;
   336→  for (let dx = 20; dx < W; dx += 120) {
   337→    ctx.globalAlpha = 0.35 + rand() * 0.3;
   338→    ctx.fillRect(dx, (roadTop + roadBot) / 2 - 3, 52, 5);
   339→  }
   340→  ctx.globalAlpha = 1;
   341→  // Potholes + tar patches + cracks.
   342→  for (let p = 0; p < 16; p++) {
   343→    const px = rand() * W;
   344→    const py = roadTop + 8 + rand() * (roadBot - roadTop - 16);
   345→    ctx.fillStyle = "rgba(24,26,28,0.55)";
   346→    ctx.beginPath();
   347→    ctx.ellipse(px, py, 7 + rand() * 16, 3 + rand() * 6, 0, 0, Math.PI * 2);
   348→    ctx.fill();
   349→  }
   350→  ctx.strokeStyle = "rgba(20,22,24,0.5)";
   351→  ctx.lineWidth = 1.2;
   352→  ctx.beginPath();
   353→  for (let k = 0; k < 26; k++) {
   354→    let cx = rand() * W;
   355→    let cy = roadTop + rand() * (roadBot - roadTop);
   356→    ctx.moveTo(cx, cy);
   357→    for (let s = 0; s < 4; s++) {
   358→      cx += (rand() - 0.5) * 34;
   359→      cy += (rand() - 0.5) * 12;
   360→      ctx.lineTo(cx, cy);
   361→    }
   362→  }
   363→  ctx.stroke();
   364→
   365→  // Dirt footpath: from the road up to the restaurant door, plus a rim path.
   366→  const pathW = 46;
   367→  const drawPath = (x0: number, y0: number, x1: number, y1: number): void => {
   368→    const steps = 22;
   369→    ctx.fillStyle = "rgba(138,119,88,0.85)";
   370→    for (let s = 0; s <= steps; s++) {
   371→      const t = s / steps;
   372→      const px = x0 + (x1 - x0) * t + Math.sin(t * 5) * 14;
   373→      const py = y0 + (y1 - y0) * t;
   374→      const w = pathW * (0.75 + 0.25 * Math.sin(t * 9));
   375→      ctx.beginPath();
   376→      ctx.ellipse(px, py, w / 2, (w / 2) * TILT * 0.9, 0, 0, Math.PI * 2);
   377→      ctx.fill();
   378→    }
   379→  };
   380→  drawPath(800, roadBot + 6, 800, localY(492)); // road → restaurant front
   381→  drawPath(120, localY(300), 380, localY(320)); // rim path upper-left
   382→  drawPath(1420, localY(320), 1560, localY(350));
   383→  drawPath(240, roadTop - 4, 500, localY(500)); // road → upper park
   384→  // Path pebbles.
   385→  for (let p = 0; p < 90; p++) {
   386→    const t = rand();
   387→    const px = 800 + Math.sin(t * 5) * 14 + (rand() - 0.5) * pathW;
   388→    const py = localY(492) + t * (roadBot + 6 - localY(492));
   389→    ctx.fillStyle = "rgba(160,144,116,0.5)";
   390→    ctx.fillRect(px, py, 2 + rand() * 2, 1.5);
   391→  }
   392→
   393→  // Grass tufts — three quick strokes each.
   394→  const tuftColors = ["#5a6440", "#6e7a4a", "#49542f", "#77824e"];
   395→  for (let t = 0; t < 620; t++) {
   396→    const px = rand() * W;
   397→    const py = rand() * H;
   398→    if (py > roadTop - 4 && py < roadBot + 4) continue;
   399→    ctx.strokeStyle = tuftColors[(rand() * tuftColors.length) | 0];
   400→    ctx.lineWidth = 1.3;
   401→    ctx.beginPath();
   402→    for (let b = 0; b < 3; b++) {
   403→      const bx = px + (b - 1) * 2.4;
   404→      ctx.moveTo(bx, py);
   405→      ctx.lineTo(bx + (rand() - 0.5) * 4, py - 4 - rand() * 6);
   406→    }
   407→    ctx.stroke();
   408→  }
   409→  // Fallen leaves / litter dots.
   410→  for (let t = 0; t < 240; t++) {
   411→    const px = rand() * W;
   412→    const py = rand() * H;
   413→    ctx.fillStyle = rand() < 0.5 ? "rgba(138,122,74,0.6)" : "rgba(90,84,52,0.55)";
   414→    ctx.fillRect(px, py, 2.4, 1.6);
   415→  }
   416→  // Small stones.
   417→  for (let t = 0; t < 120; t++) {
   418→    const px = rand() * W;
   419→    const py = rand() * H;
   420→    const s = 1.6 + rand() * 3.4;
   421→    ctx.fillStyle = "#6c675c";
   422→    ctx.beginPath();
   423→    ctx.ellipse(px, py, s, s * 0.62, 0, 0, Math.PI * 2);
   424→    ctx.fill();
   425→    ctx.fillStyle = "rgba(190,182,166,0.5)";
   426→    ctx.fillRect(px - s * 0.4, py - s * 0.5, s * 0.7, s * 0.3);
   427→  }
   428→
   429→  // Arena boundary: weathered concrete curb + short barrier posts.
   430→  ctx.fillStyle = "#57544c";
   431→  ctx.fillRect(0, 0, W, 5); // far curb (horizon edge)
   432→  ctx.fillRect(0, H - 6, W, 6); // near curb
   433→  ctx.fillStyle = "#454239";
   434→  for (let px = 8; px < W; px += 88) {
   435→    ctx.fillRect(px, H - 20, 7, 15); // near posts
   436→    ctx.fillStyle = "#6a675e";
   437→    ctx.fillRect(px + 1, H - 20, 2, 15);
   438→    ctx.fillStyle = "#454239";
   439→  }
   440→  // Chain-link fence hint along the horizon.
   441→  ctx.strokeStyle = "rgba(70,66,60,0.85)";
   442→  ctx.lineWidth = 1.6;
   443→  ctx.beginPath();
   444→  ctx.moveTo(0, 7);
   445→  ctx.lineTo(W, 7);
   446→  ctx.stroke();
   447→  ctx.strokeStyle = "rgba(90,86,78,0.5)";
   448→  ctx.lineWidth = 1;
   449→  ctx.beginPath();
   450→  for (let px = 0; px < W; px += 10) {
   451→    ctx.moveTo(px, 2);
   452→    ctx.lineTo(px + 6, 9);
   453→    ctx.moveTo(px + 6, 2);
   454→    ctx.lineTo(px, 9);
   455→  }
   456→  ctx.stroke();
   457→
   458→  // Corner vignette so the arena edges melt into the frame.
   459→  const vg = ctx.createRadialGradient(W / 2, H * 0.42, H * 0.4, W / 2, H * 0.5, H * 1.05);
   460→  vg.addColorStop(0, "rgba(0,0,0,0)");
   461→  vg.addColorStop(1, "rgba(8,10,6,0.34)");
   462→  ctx.fillStyle = vg;
   463→  ctx.fillRect(0, 0, W, H);
   464→  return c;

   475→function brickWallFace(ctx: CanvasRenderingContext2D, w: number, h: number, seed: number): void {
   476→  const rand = rng32(seed);
   477→  // Course gradient.

   820→function bakeStructure(index: number): HTMLCanvasElement {
   821→  const b = OBSTACLES[index];
   822→  const H = STRUCT_HEIGHTS[index];
   823→  const [c, ctx] = makeCanvas(b.w, H);
   824→  switch (index) {
   825→    case 0:
   826→      brickWallFace(ctx, b.w, H, 0xB100 + 7);
   827→      break;
   828→    case 1:
   829→      brickWallFace(ctx, b.w, H, 0xB200 + 11);
   830→      break;
   831→    case 2:
   832→      kioskFace(ctx, b.w, H, 0xC200 + 3);
   833→      break;
   834→    case 3:
   835→      restaurantFace(ctx, b.w, H, 0xD300 + 5);
   836→      break;
   837→    default:
   838→      cabinFace(ctx, b.w, H, 0xE400 + 9);
   839→  }
   840→  return c;
   841→}

   926→      const rx = b.x + 20 + rand() * (b.w - 40);
   927→      const rh = 6 + rand() * 10;
   928→      ctx.moveTo(rx, yFar - H);
   929→      ctx.lineTo(rx + 2, yFar - H - rh);
   930→    }
   931→    ctx.stroke();
   932→  }
   933→  // Outline for readability.
   934→  ctx.strokeStyle = "rgba(15,12,10,0.55)";
   935→  ctx.lineWidth = 1.4;
   936→  ctx.strokeRect(b.x, yFar - H, b.w, topH + H);
   937→}

   941→function drawTree(ctx: CanvasRenderingContext2D, d: Decor, t: number): void {
   942→  const ax = d.x;
   943→  const ay = ENV_HORIZON + localY(d.y);
   944→  const S = 52 * d.scale;
   945→  const sway = Math.sin(t * 1.1 + d.variant * 1.7) * 0.028 + Math.sin(t * 2.3 + d.variant) * 0.012;
   946→
   947→  // Shadow.
   948→  ctx.fillStyle = "rgba(10,12,8,0.4)";
   949→  ctx.beginPath();
   950→  ctx.ellipse(ax + 6, ay + 3, S * 0.55, S * 0.14, 0, 0, Math.PI * 2);
   951→  ctx.fill();
   952→
   953→  const lean = sway * S;
   954→  if (d.dead) {
   955→    // Bare dead tree — recursive-ish branches.
   956→    ctx.strokeStyle = C.deadWood;
   957→    ctx.lineCap = "round";
   958→    ctx.lineWidth = S * 0.09;
   959→    ctx.beginPath();
   960→    ctx.moveTo(ax, ay);
   961→    const topX = ax + lean;
   962→    const topY = ay - S * 1.75;
   963→    ctx.quadraticCurveTo(ax + S * 0.06, ay - S * 0.9, topX, topY);
   964→    ctx.stroke();
   965→    const branch = (
   966→      x0: number, y0: number, ang: number, len: number, w: number, depth: number,
   967→    ): void => {
   968→      if (depth === 0 || len < 4) return;
   969→      const x1 = x0 + Math.cos(ang) * len;
   970→      const y1 = y0 + Math.sin(ang) * len;
   971→      ctx.lineWidth = w;
   972→      ctx.beginPath();
   973→      ctx.moveTo(x0, y0);
   974→      ctx.lineTo(x1, y1);
   975→      ctx.stroke();
   976→      branch(x1, y1, ang - 0.42 - sway * 0.4, len * 0.68, w * 0.6, depth - 1);
   977→      branch(x1, y1, ang + 0.38 + sway * 0.4, len * 0.66, w * 0.6, depth - 1);
   978→    };
   979→    const tipX = ax + lean;
   980→    const tipY = ay - S * 1.75;
   981→    branch(tipX, tipY + S * 0.35, -Math.PI / 2 - 0.65 + sway, S * 0.5, S * 0.06, 3);
   982→    branch(tipX, tipY + S * 0.55, -Math.PI / 2 + 0.6 + sway, S * 0.55, S * 0.07, 3);
   983→    branch(tipX, tipY + S * 0.15, -Math.PI / 2 + 0.18 + sway * 0.5, S * 0.4, S * 0.05, 2);
   984→    return;
   985→  }
   986→
   987→  // Living tree: trunk + layered canopy blobs.
   988→  ctx.strokeStyle = "#3e332a";
   989→  ctx.lineCap = "round";
   990→  ctx.lineWidth = S * 0.13;
   991→  ctx.beginPath();
   992→  ctx.moveTo(ax, ay);
   993→  ctx.quadraticCurveTo(ax + S * 0.05, ay - S * 0.8, ax + lean, ay - S * 1.35);
   994→  ctx.stroke();
   995→  // Branch stubs.
   996→  ctx.lineWidth = S * 0.06;
   997→  ctx.beginPath();
   998→  ctx.moveTo(ax + S * 0.02, ay - S * 0.85);
   999→  ctx.lineTo(ax - S * 0.34 + lean * 0.6, ay - S * 1.15);
  1000→  ctx.moveTo(ax + S * 0.03, ay - S * 1.0);
  1001→  ctx.lineTo(ax + S * 0.38 + lean * 0.6, ay - S * 1.28);
  1002→  ctx.stroke();
  1003→
  1004→  const cy = ay - S * 1.62;
  1005→  const cx = ax + lean * 1.6;
  1006→  const blobs: [number, number, number][] = [
  1007→    [0, 0, 0.62],
  1008→    [-0.42, 0.16, 0.44],
  1009→    [0.44, 0.12, 0.46],
  1010→    [-0.18, -0.3, 0.4],
  1011→    [0.24, -0.26, 0.38],
  1012→  ];
  1013→  for (const [ox, oy, r] of blobs) {
  1014→    ctx.fillStyle = C.leafDark;
  1015→    ctx.beginPath();
  1016→    ctx.ellipse(cx + ox * S, cy + oy * S, r * S, r * S * 0.86, 0, 0, Math.PI * 2);
  1017→    ctx.fill();
  1018→  }
  1019→  for (const [ox, oy, r] of blobs) {
  1020→    ctx.fillStyle = C.leaf;
  1021→    ctx.beginPath();
  1022→    ctx.ellipse(cx + ox * S - 1.5, cy + oy * S - 2.5, r * S * 0.9, r * S * 0.76, 0, 0, Math.PI * 2);
  1023→    ctx.fill();
  1024→  }
  1025→  // Sun-side highlights.
  1026→  for (const [ox, oy, r] of blobs) {
  1027→    ctx.fillStyle = C.leafLight;
  1028→    ctx.beginPath();
  1029→    ctx.ellipse(cx + ox * S + r * S * 0.24, cy + oy * S - r * S * 0.3, r * S * 0.42, r * S * 0.3, 0, 0, Math.PI * 2);
  1030→    ctx.fill();
  1031→  }
  1032→}

  1034→function drawBush(ctx: CanvasRenderingContext2D, d: Decor, t: number): void {
  1035→  const ax = d.x;
  1036→  const ay = ENV_HORIZON + localY(d.y);
  1037→  const S = 20 * d.scale;
  1038→  const sway = Math.sin(t * 1.6 + d.variant * 2.1) * 0.05;
  1039→
  1040→  ctx.fillStyle = "rgba(10,12,8,0.35)";
  1041→  ctx.beginPath();
  1042→  ctx.ellipse(ax, ay + 2, S * 1.1, S * 0.24, 0, 0, Math.PI * 2);
  1043→  ctx.fill();
  1044→
  1045→  const blobs: [number, number, number][] = [
  1046→    [-0.6, -0.2, 0.62],
  1047→    [0.55, -0.15, 0.58],
  1048→    [0, -0.5, 0.66],
  1049→  ];
  1050→  for (const [ox, oy, r] of blobs) {
  1051→    ctx.fillStyle = C.leafDark;
  1052→    ctx.beginPath();
  1053→    ctx.ellipse(ax + ox * S * (1 + sway), ay - oy * S, r * S, r * S * 0.78, 0, 0, Math.PI * 2);
  1054→    ctx.fill();
  1055→  }
  1056→  for (const [ox, oy, r] of blobs) {
  1057→    ctx.fillStyle = C.leaf;
  1058→    ctx.beginPath();
  1059→    ctx.ellipse(ax + ox * S * (1 + sway) - 1, ay - oy * S - 1.5, r * S * 0.82, r * S * 0.6, 0, 0, Math.PI * 2);
  1060→    ctx.fill();
  1061→  }
  1062→  ctx.fillStyle = C.leafLight;
  1063→  ctx.beginPath();
  1064→  ctx.ellipse(ax + S * 0.18, ay - S * 0.72, S * 0.34, S * 0.22, 0, 0, Math.PI * 2);
  1065→  ctx.fill();
  1066→}

  1068→function drawRock(ctx: CanvasRenderingContext2D, d: Decor): void {
  1069→  const ax = d.x;
  1070→  const ay = ENV_HORIZON + localY(d.y);
  1071→  const S = 9 * d.scale;
  1072→  ctx.fillStyle = "rgba(10,12,8,0.35)";
  1073→  ctx.beginPath();
  1074→  ctx.ellipse(ax + 1, ay + 1.5, S * 1.15, S * 0.3, 0, 0, Math.PI * 2);
  1075→  ctx.fill();
  1076→  ctx.fillStyle = "#6c675c";
  1077→  ctx.beginPath();
  1078→  ctx.moveTo(ax - S, ay);
  1079→  ctx.lineTo(ax - S * 0.6, ay - S * 0.85);
  1080→  ctx.lineTo(ax + S * 0.2, ay - S);
  1081→  ctx.lineTo(ax + S, ay - S * 0.35);
  1082→  ctx.lineTo(ax + S * 0.9, ay);
  1083→  ctx.closePath();
  1084→  ctx.fill();
  1085→  ctx.fillStyle = "#847e70";
  1086→  ctx.beginPath();
  1087→  ctx.moveTo(ax - S * 0.6, ay - S * 0.85);
  1088→  ctx.lineTo(ax + S * 0.2, ay - S);
  1089→  ctx.lineTo(ax + S * 0.3, ay - S * 0.5);
  1090→  ctx.closePath();
  1091→  ctx.fill();
  1092→}

  1095→export function drawDecor(ctx: CanvasRenderingContext2D, d: Decor, t: number): void {
  1096→  if (d.kind === "tree") drawTree(ctx, d, t);
  1097→  else if (d.kind === "bush") drawBush(ctx, d, t);
  1098→  else drawRock(ctx, d);

  1103→interface Cloud {
  1104→  x: number;
  1105→  y: number;
  1106→  s: number;
  1107→  v: number;
  1108→}
  1109→let clouds: Cloud[] | null = null;

  1111→function getClouds(): Cloud[] {
  1112→  if (clouds) return clouds;
  1113→  const rand = rng32(0xC10D);
  1114→  clouds = Array.from({ length: 6 }, () => ({
  1115→    x: rand() * WORLD_W,
  1116→    y: 14 + rand() * 120,
  1117→    s: 0.6 + rand() * 1.1,
  1118→    v: 4 + rand() * 7,
  1119→  }));
  1120→  return clouds;
  1121→}

  1123→function drawCloud(ctx: CanvasRenderingContext2D, c: Cloud, t: number): void {
  1124→  const x = ((c.x + t * c.v) % (WORLD_W + 260)) - 130;
  1125→  const y = c.y;
  1126→  const s = c.s;
  1127→  ctx.fillStyle = "rgba(160,138,118,0.14)";
  1128→  ctx.beginPath();
  1129→  ctx.ellipse(x, y, 60 * s, 14 * s, 0, 0, Math.PI * 2);
  1130→  ctx.ellipse(x + 34 * s, y - 8 * s, 40 * s, 12 * s, 0, 0, Math.PI * 2);
  1131→  ctx.ellipse(x - 36 * s, y + 3 * s, 34 * s, 10 * s, 0, 0, Math.PI * 2);
  1132→  ctx.fill();
  1133→}

  1135→function drawBirds(ctx: CanvasRenderingContext2D, t: number): void {
  1136→  ctx.strokeStyle = "rgba(30,24,20,0.7)";
  1137→  ctx.lineWidth = 1.6;
  1138→  for (let b = 0; b < 5; b++) {
  1139→    const bx = ((b * 173 + t * (16 + b * 3)) % (WORLD_W + 140)) - 70;
  1140→    const by = 44 + b * 13 + Math.sin(t * 2 + b) * 5;
  1141→    const flap = Math.sin(t * 7 + b * 1.9) * 3;
  1142→    ctx.beginPath();
  1143→    ctx.moveTo(bx - 5, by - flap);
  1144→    ctx.quadraticCurveTo(bx, by + 2, bx + 5, by - flap);
  1145→    ctx.stroke();
  1146→  }
  1147→}
